/**
 * Get all categories
 * @returns {Array} Array of category objects
 */
export function getCategories() {
  try {
    const categories = storage.get(CATEGORIES_KEY, []);
    
    // If no categories exist, initialize with defaults
    if (categories.length === 0) {
      return initializeDefaultCategories();
    }
    
    return categories.sort((a, b) => {
      // Sort default categories first, then by name
      if (a.isDefault && !b.isDefault) return -1;
      if (!a.isDefault && b.isDefault) return 1;
      return a.name.localeCompare(b.name);
    });
  } catch (error) {
    console.error('Error getting categories:', error);
    return initializeDefaultCategories();
  }
}

/**
 * Initialize default categories
 * @returns {Array} Array of default categories
 */
export function initializeDefaultCategories() {
  try {
    const categoriesWithTimestamps = DEFAULT_CATEGORIES.map(category => ({
      ...category,
      createdAt: new Date().toISOString()
    }));

    const success = storage.set(CATEGORIES_KEY, categoriesWithTimestamps);
    if (!success) {
      throw new Error('Failed to initialize default categories');
    }

    return categoriesWithTimestamps;
  } catch (error) {
    console.error('Error initializing default categories:', error);
    // Return default categories even if storage fails
    return DEFAULT_CATEGORIES.map(category => ({
      ...category,
      createdAt: new Date().toISOString()
    }));
  }
}

/**
 * Add a new custom category
 * @param {Object} categoryData - Category data object
 * @returns {Object} Created category with generated ID
 * @throws {Error} If validation fails or storage error occurs
 */
export function addCategory(categoryData) {
  try {
    // Validate category data
    const validatedCategory = validateCategory(categoryData);
    
    // Check for duplicate names
    const existingCategories = getCategories();
    const nameExists = existingCategories.some(
      cat => cat.name.toLowerCase() === validatedCategory.name.toLowerCase()
    );
    
    if (nameExists) {
      throw new Error(`Category with name "${validatedCategory.name}" already exists`);
    }

    // Generate unique ID and timestamps
    const category = {
      ...validatedCategory,
      id: categoryData.id || generateCategoryId(validatedCategory.name),
      isDefault: false, // Custom categories are never default
      createdAt: new Date().toISOString()
    };

    // Add to existing categories
    const categories = existingCategories;
    categories.push(category);

    // Save to storage
    const success = storage.set(CATEGORIES_KEY, categories);
    if (!success) {
      throw new Error('Failed to save category to storage');
    }

    return category;
  } catch (error) {
    console.error('Error adding category:', error);
    throw new Error(`Failed to add category: ${error.message}`);
  }
}

/**
 * Update an existing category
 * @param {string} id - Category ID
 * @param {Object} updateData - Data to update
 * @returns {Object} Updated category object
 * @throws {Error} If category not found or validation fails
 */
export function updateCategory(id, updateData) {
  try {
    const categories = getCategories();
    const categoryIndex = categories.findIndex(category => category.id === id);
    
    if (categoryIndex === -1) {
      throw new Error(`Category with ID ${id} not found`);
    }

    const existingCategory = categories[categoryIndex];
    
    // Prevent updating default categories' core properties
    if (existingCategory.isDefault) {
      const allowedUpdates = ['name', 'color', 'icon'];
      const hasDisallowedUpdate = Object.keys(updateData).some(
        key => !allowedUpdates.includes(key)
      );
      
      if (hasDisallowedUpdate) {
        throw new Error('Cannot modify core properties of default categories');
      }
    }

    // Check for duplicate names (excluding current category)
    if (updateData.name) {
      const nameExists = categories.some(
        cat => cat.id !== id && 
        cat.name.toLowerCase() === updateData.name.toLowerCase()
      );
      
      if (nameExists) {
        throw new Error(`Category with name "${updateData.name}" already exists`);
      }
    }

    // Merge update data with existing category
    const updatedCategoryData = {
      ...existingCategory,
      ...updateData,
      id: existingCategory.id, // Preserve original ID
      isDefault: existingCategory.isDefault, // Preserve default status
      createdAt: existingCategory.createdAt // Preserve creation date
    };

    // Validate updated category
    const validatedCategory = validateCategory(updatedCategoryData);

    // Update in array
    categories[categoryIndex] = validatedCategory;

    // Save to storage
    const success = storage.set(CATEGORIES_KEY, categories);
    if (!success) {
      throw new Error('Failed to save updated category to storage');
    }

    return validatedCategory;
  } catch (error) {
    console.error('Error updating category:', error);
    throw new Error(`Failed to update category: ${error.message}`);
  }
}

/**
 * Delete a category by ID
 * @param {string} id - Category ID
 * @param {string} reassignToCategoryId - ID of category to reassign expenses to
 * @returns {boolean} Success status
 * @throws {Error} If category not found or is default category
 */
export function deleteCategory(id, reassignToCategoryId = 'other') {
  try {
    const categories = getCategories();
    const categoryToDelete = categories.find(cat => cat.id === id);
    
    if (!categoryToDelete) {
      throw new Error(`Category with ID ${id} not found`);
    }

    // Prevent deletion of default categories
    if (categoryToDelete.isDefault) {
      throw new Error('Cannot delete default categories');
    }

    // Verify reassign category exists
    const reassignCategory = categories.find(cat => cat.id === reassignToCategoryId);
    if (!reassignCategory) {
      throw new Error(`Reassign category with ID ${reassignToCategoryId} not found`);
    }

    // Reassign expenses to new category
    await reassignExpensesToCategory(id, reassignToCategoryId);

    // Remove category from array
    const filteredCategories = categories.filter(category => category.id !== id);

    // Save updated categories
    const success = storage.set(CATEGORIES_KEY, filteredCategories);
    if (!success) {
      throw new Error('Failed to save categories after deletion');
    }

    return true;
  } catch (error) {
    console.error('Error deleting category:', error);
    throw new Error(`Failed to delete category: ${error.message}`);
  }
}

/**
 * Get a category by ID
 * @param {string} id - Category ID
 * @returns {Object|null} Category object or null if not found
 */
export function getCategoryById(id) {
  try {
    const categories = getCategories();
    return categories.find(category => category.id === id) || null;
  } catch (error) {
    console.error('Error getting category by ID:', error);
    return null;
  }
}

/**
 * Get only custom categories (non-default)
 * @returns {Array} Array of custom categories
 */
export function getCustomCategories() {
  try {
    const categories = getCategories();
    return categories.filter(category => !category.isDefault);
  } catch (error) {
    console.error('Error getting custom categories:', error);
    return [];
  }
}

/**
 * Get only default categories
 * @returns {Array} Array of default categories
 */
export function getDefaultCategories() {
  try {
    const categories = getCategories();
    return categories.filter(category => category.isDefault);
  } catch (error) {
    console.error('Error getting default categories:', error);
    return DEFAULT_CATEGORIES;
  }
}

/**
 * Reassign all expenses from one category to another
 * @param {string} fromCategoryId - Source category ID
 * @param {string} toCategoryId - Target category ID
 * @returns {number} Number of expenses reassigned
 */
async function reassignExpensesToCategory(fromCategoryId, toCategoryId) {
  try {
    // Import here to avoid circular dependency
    const { getAllExpenses, updateExpense } = await import('./expenseStorage.js');
    
    const expenses = getAllExpenses();
    const expensesToReassign = expenses.filter(
      expense => expense.categoryId === fromCategoryId
    );

    // Update each expense
    let reassignedCount = 0;
    for (const expense of expensesToReassign) {
      try {
        await updateExpense(expense.id, { categoryId: toCategoryId });
        reassignedCount++;
      } catch (error) {
        console.error(`Failed to reassign expense ${expense.id}:`, error);
      }
    }

    return reassignedCount;
  } catch (error) {
    console.error('Error reassigning expenses:', error);
    return 0;
  }
}

/**
 * Get category usage statistics
 * @returns {Object} Usage statistics for each category
 */
export function getCategoryUsageStats() {
  try {
    // Import here to avoid circular dependency
    const { getAllExpenses } = require('./expenseStorage.js');
    
    const expenses = getAllExpenses();
    const categories = getCategories();
    
    const usageStats = categories.map(category => {
      const categoryExpenses = expenses.filter(
        expense => expense.categoryId === category.id
      );
      
      const total = categoryExpenses.reduce(
        (sum, expense) => sum + expense.amount, 0
      );
      
      return {
        categoryId: category.id,
        name: category.name,
        color: category.color,
        expenseCount: categoryExpenses.length,
        totalAmount: total,
        percentage: expenses.length > 0 ? 
          Math.round((categoryExpenses.length / expenses.length) * 100) : 0
      };
    });

    return usageStats.sort((a, b) => b.totalAmount - a.totalAmount);
  } catch (error) {
    console.error('Error getting category usage stats:', error);
    return [];
  }
}

/**
 * Reset categories to defaults (remove all custom categories)
 * @returns {boolean} Success status
 */
export function resetCategoriesToDefaults() {
  try {
    // Get current custom categories
    const customCategories = getCustomCategories();
    
    // Reassign all expenses from custom categories to 'other'
    const reassignPromises = customCategories.map(category =>
      reassignExpensesToCategory(category.id, 'other')
    );
    
    Promise.all(reassignPromises);

    // Initialize default categories
    const defaultCategories = initializeDefaultCategories();
    
    return true;
  } catch (error) {
    console.error('Error resetting categories:', error);
    return false;
  }
}

/**
 * Export categories data for backup
 * @returns {Object} Categories data with metadata
 */
export function exportCategoriesData() {
  try {
    const categories = getCategories();
    
    return {
      categories,
      exportDate: new Date().toISOString(),
      version: '1.0',
      type: 'categories'
    };
  } catch (error) {
    console.error('Error exporting categories data:', error);
    return null;
  }
}

/**
 * Import categories data from backup
 * @param {Object} data - Exported categories data
 * @returns {boolean} Success status
 */
export function importCategoriesData(data) {
  try {
    if (!data || !data.categories || !Array.isArray(data.categories)) {
      throw new Error('Invalid categories data format');
    }

    // Validate each category
    const validatedCategories = data.categories.map(category => {
      return validateCategory(category);
    });

    // Save to storage
    const success = storage.set(CATEGORIES_KEY, validatedCategories);
    if (!success) {
      throw new Error('Failed to save imported categories');
    }

    return true;
  } catch (error) {
    console.error('Error importing categories data:', error);
    return false;
  }
}
