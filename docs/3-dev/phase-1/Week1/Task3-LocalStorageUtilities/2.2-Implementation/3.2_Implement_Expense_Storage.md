### Expense Storage Utilities (`src/utils/expenseStorage.js`)

```javascript
/**
 * Expense data management utilities
 */

import { storage, StorageError } from './storage.js';
import { validateExpense, generateExpenseId } from './validation.js';

const EXPENSES_KEY = 'expenses';

/**
 * Get all expenses sorted by date (newest first)
 * @returns {Array} Array of expense objects
 */
export function getAllExpenses() {
  try {
    const expenses = storage.get(EXPENSES_KEY, []);
    return expenses.sort((a, b) => new Date(b.date) - new Date(a.date));
  } catch (error) {
    console.error('Error getting all expenses:', error);
    return [];
  }
}

/**
 * Add a new expense
 * @param {Object} expenseData - Expense data object
 * @returns {Object} Created expense with generated ID
 * @throws {Error} If validation fails or storage error occurs
 */
export function addExpense(expenseData) {
  try {
    // Validate expense data
    const validatedExpense = validateExpense(expenseData);
    
    // Generate unique ID and timestamps
    const expense = {
      ...validatedExpense,
      id: generateExpenseId(),
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString()
    };

    // Get current expenses and add new one
    const expenses = getAllExpenses();
    expenses.push(expense);

    // Save to storage
    const success = storage.set(EXPENSES_KEY, expenses);
    if (!success) {
      throw new Error('Failed to save expense to storage');
    }

    return expense;
  } catch (error) {
    console.error('Error adding expense:', error);
    throw new Error(`Failed to add expense: ${error.message}`);
  }
}

/**
 * Update an existing expense
 * @param {string} id - Expense ID
 * @param {Object} updateData - Data to update
 * @returns {Object} Updated expense object
 * @throws {Error} If expense not found or validation fails
 */
export function updateExpense(id, updateData) {
  try {
    const expenses = getAllExpenses();
    const expenseIndex = expenses.findIndex(expense => expense.id === id);
    
    if (expenseIndex === -1) {
      throw new Error(`Expense with ID ${id} not found`);
    }

    // Merge update data with existing expense
    const existingExpense = expenses[expenseIndex];
    const updatedExpenseData = {
      ...existingExpense,
      ...updateData,
      id: existingExpense.id, // Preserve original ID
      createdAt: existingExpense.createdAt, // Preserve creation date
      updatedAt: new Date().toISOString()
    };

    // Validate updated expense
    const validatedExpense = validateExpense(updatedExpenseData);

    // Update in array
    expenses[expenseIndex] = validatedExpense;

    // Save to storage
    const success = storage.set(EXPENSES_KEY, expenses);
    if (!success) {
      throw new Error('Failed to save updated expense to storage');
    }

    return validatedExpense;
  } catch (error) {
    console.error('Error updating expense:', error);
    throw new Error(`Failed to update expense: ${error.message}`);
  }
}

/**
 * Delete an expense by ID
 * @param {string} id - Expense ID
 * @returns {boolean} Success status
 * @throws {Error} If expense not found
 */
export function deleteExpense(id) {
  try {
    const expenses = getAllExpenses();
    const initialLength = expenses.length;
    const filteredExpenses = expenses.filter(expense => expense.id !== id);
    
    if (filteredExpenses.length === initialLength) {
      throw new Error(`Expense with ID ${id} not found`);
    }

    // Save filtered expenses
    const success = storage.set(EXPENSES_KEY, filteredExpenses);
    if (!success) {
      throw new Error('Failed to save expenses after deletion');
    }

    return true;
  } catch (error) {
    console.error('Error deleting expense:', error);
    throw new Error(`Failed to delete expense: ${error.message}`);
  }
}

/**
 * Get expenses for a specific date
 * @param {string} date - Date string in YYYY-MM-DD format
 * @returns {Array} Array of expenses for the date
 */
export function getExpensesByDate(date) {
  try {
    const expenses = getAllExpenses();
    return expenses.filter(expense => expense.date === date);
  } catch (error) {
    console.error('Error getting expenses by date:', error);
    return [];
  }
}

/**
 * Get expenses for a specific month
 * @param {number} year - Year (e.g., 2024)
 * @param {number} month - Month (1-12)
 * @returns {Array} Array of expenses for the month
 */
export function getExpensesByMonth(year, month) {
  try {
    const expenses = getAllExpenses();
    const monthStr = month.toString().padStart(2, '0');
    const yearMonthPrefix = `${year}-${monthStr}`;
    
    return expenses.filter(expense => 
      expense.date.startsWith(yearMonthPrefix)
    );
  } catch (error) {
    console.error('Error getting expenses by month:', error);
    return [];
  }
}

/**
 * Get expenses within a date range
 * @param {string} startDate - Start date (YYYY-MM-DD)
 * @param {string} endDate - End date (YYYY-MM-DD)
 * @returns {Array} Array of expenses in range
 */
export function getExpensesByDateRange(startDate, endDate) {
  try {
    const expenses = getAllExpenses();
    return expenses.filter(expense => {
      const expenseDate = expense.date;
      return expenseDate >= startDate && expenseDate <= endDate;
    });
  } catch (error) {
    console.error('Error getting expenses by date range:', error);
    return [];
  }
}

/**
 * Get expenses by category
 * @param {string} categoryId - Category ID
 * @returns {Array} Array of expenses for the category
 */
export function getExpensesByCategory(categoryId) {
  try {
    const expenses = getAllExpenses();
    return expenses.filter(expense => expense.categoryId === categoryId);
  } catch (error) {
    console.error('Error getting expenses by category:', error);
    return [];
  }
}

/**
 * Calculate total expenses for a date
 * @param {string} date - Date string (YYYY-MM-DD)
 * @returns {number} Total amount for the date
 */
export function getDailyTotal(date) {
  try {
    const dayExpenses = getExpensesByDate(date);
    return dayExpenses.reduce((total, expense) => total + expense.amount, 0);
  } catch (error) {
    console.error('Error calculating daily total:', error);
    return 0;
  }
}

/**
 * Calculate total expenses for a month
 * @param {number} year - Year
 * @param {number} month - Month (1-12)
 * @returns {number} Total amount for the month
 */
export function getMonthlyTotal(year, month) {
  try {
    const monthExpenses = getExpensesByMonth(year, month);
    return monthExpenses.reduce((total, expense) => total + expense.amount, 0);
  } catch (error) {
    console.error('Error calculating monthly total:', error);
    return 0;
  }
}

/**
 * Get expense statistics for a period
 * @param {Array} expenses - Array of expenses
 * @returns {Object} Statistics object
 */
export function getExpenseStatistics(expenses) {
  try {
    if (!expenses || expenses.length === 0) {
      return {
        total: 0,
        count: 0,
        average: 0,
        highest: 0,
        lowest: 0,
        byCategory: {}
      };
    }

    const amounts = expenses.map(e => e.amount);
    const total = amounts.reduce((sum, amount) => sum + amount, 0);
    
    // Group by category
    const byCategory = expenses.reduce((acc, expense) => {
      const categoryId = expense.categoryId;
      if (!acc[categoryId]) {
        acc[categoryId] = { total: 0, count: 0 };
      }
      acc[categoryId].total += expense.amount;
      acc[categoryId].count += 1;
      return acc;
    }, {});

    return {
      total,
      count: expenses.length,
      average: total / expenses.length,
      highest: Math.max(...amounts),
      lowest: Math.min(...amounts),
      byCategory
    };
  } catch (error) {
    console.error('Error calculating expense statistics:', error);
    return {
      total: 0,
      count: 0,
      average: 0,
      highest: 0,
      lowest: 0,
      byCategory: {}
    };
  }
}

/**
 * Search expenses by description
 * @param {string} query - Search query
 * @returns {Array} Matching expenses
 */
export function searchExpenses(query) {
  try {
    if (!query || query.trim() === '') {
      return [];
    }

    const expenses = getAllExpenses();
    const searchTerm = query.toLowerCase().trim();
    
    return expenses.filter(expense => {
      const description = expense.description || '';
      return description.toLowerCase().includes(searchTerm);
    });
  } catch (error) {
    console.error('Error searching expenses:', error);
    return [];
  }
}

/**
 * Get recent expenses (last N days)
 * @param {number} days - Number of days to look back
 * @returns {Array} Recent expenses
 */
export function getRecentExpenses(days = 7) {
  try {
    const expenses = getAllExpenses();
    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - days);
    const cutoffStr = cutoffDate.toISOString().split('T')[0];
    
    return expenses.filter(expense => expense.date >= cutoffStr);
  } catch (error) {
    console.error('Error getting recent expenses:', error);
    return [];
  }
}
```
