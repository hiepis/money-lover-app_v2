### Core Storage Utility (`src/utils/storage.js`)

```javascript
/**
 * Core localStorage utility with safe JSON operations and error handling
 */

class StorageError extends Error {
  constructor(message, operation, key) {
    super(message);
    this.name = 'StorageError';
    this.operation = operation;
    this.key = key;
  }
}

export const storage = {
  /**
   * Safely get and parse data from localStorage
   * @param {string} key - Storage key
   * @param {*} defaultValue - Default value if key doesn't exist
   * @returns {*} Parsed data or default value
   */
  get: (key, defaultValue = null) => {
    try {
      if (!isStorageAvailable()) {
        console.warn('localStorage not available');
        return defaultValue;
      }

      const item = localStorage.getItem(key);
      if (item === null) {
        return defaultValue;
      }

      return JSON.parse(item);
    } catch (error) {
      console.error(`Storage get error for key "${key}":`, error);
      return defaultValue;
    }
  },

  /**
   * Safely stringify and store data in localStorage
   * @param {string} key - Storage key
   * @param {*} value - Data to store
   * @returns {boolean} Success status
   */
  set: (key, value) => {
    try {
      if (!isStorageAvailable()) {
        throw new StorageError('localStorage not available', 'set', key);
      }

      const serialized = JSON.stringify(value);
      
      // Check storage quota before writing
      if (!checkStorageQuota(key, serialized)) {
        throw new StorageError('Storage quota exceeded', 'set', key);
      }

      localStorage.setItem(key, serialized);
      return true;
    } catch (error) {
      if (error.name === 'QuotaExceededError') {
        console.error('Storage quota exceeded');
        throw new StorageError('Storage quota exceeded', 'set', key);
      }
      console.error(`Storage set error for key "${key}":`, error);
      throw new StorageError(`Failed to store data: ${error.message}`, 'set', key);
    }
  },

  /**
   * Remove item from localStorage
   * @param {string} key - Storage key
   * @returns {boolean} Success status
   */
  remove: (key) => {
    try {
      if (!isStorageAvailable()) {
        return false;
      }
      localStorage.removeItem(key);
      return true;
    } catch (error) {
      console.error(`Storage remove error for key "${key}":`, error);
      return false;
    }
  },

  /**
   * Clear all app data from localStorage
   * @param {string[]} appKeys - Array of app-specific keys to clear
   * @returns {boolean} Success status
   */
  clear: (appKeys = ['expenses', 'categories', 'app_settings', 'data_version']) => {
    try {
      if (!isStorageAvailable()) {
        return false;
      }
      
      appKeys.forEach(key => {
        localStorage.removeItem(key);
      });
      return true;
    } catch (error) {
      console.error('Storage clear error:', error);
      return false;
    }
  },

  /**
   * Get current storage usage
   * @returns {object} Storage usage information
   */
  getSize: () => {
    try {
      if (!isStorageAvailable()) {
        return { total: 0, available: 0, used: 0 };
      }

      let total = 0;
      for (let key in localStorage) {
        if (localStorage.hasOwnProperty(key)) {
          const value = localStorage.getItem(key);
          total += key.length + (value ? value.length : 0);
        }
      }

      return {
        used: total,
        total: getStorageQuota(),
        available: getStorageQuota() - total,
        usagePercent: Math.round((total / getStorageQuota()) * 100)
      };
    } catch (error) {
      console.error('Storage size calculation error:', error);
      return { total: 0, available: 0, used: 0, usagePercent: 0 };
    }
  },

  /**
   * Check if a key exists in localStorage
   * @param {string} key - Storage key
   * @returns {boolean} Key existence
   */
  has: (key) => {
    try {
      if (!isStorageAvailable()) {
        return false;
      }
      return localStorage.getItem(key) !== null;
    } catch (error) {
      console.error(`Storage has error for key "${key}":`, error);
      return false;
    }
  },

  /**
   * Get all keys for the app
   * @param {string} prefix - Key prefix filter
   * @returns {string[]} Array of keys
   */
  keys: (prefix = '') => {
    try {
      if (!isStorageAvailable()) {
        return [];
      }
      
      const keys = [];
      for (let i = 0; i < localStorage.length; i++) {
        const key = localStorage.key(i);
        if (key && key.startsWith(prefix)) {
          keys.push(key);
        }
      }
      return keys;
    } catch (error) {
      console.error('Storage keys error:', error);
      return [];
    }
  }
};

/**
 * Check if localStorage is available
 * @returns {boolean} Storage availability
 */
function isStorageAvailable() {
  try {
    const test = '__storage_test__';
    localStorage.setItem(test, test);
    localStorage.removeItem(test);
    return true;
  } catch (error) {
    return false;
  }
}

/**
 * Check storage quota before writing
 * @param {string} key - Storage key
 * @param {string} data - Serialized data
 * @returns {boolean} Can write status
 */
function checkStorageQuota(key, data) {
  try {
    const currentSize = storage.getSize().used;
    const newDataSize = key.length + data.length;
    const existingItemSize = localStorage.getItem(key)?.length || 0;
    const totalAfterWrite = currentSize - existingItemSize + newDataSize;
    
    // Use 80% of quota as safe limit
    const safeQuota = getStorageQuota() * 0.8;
    return totalAfterWrite <= safeQuota;
  } catch (error) {
    console.warn('Could not check storage quota:', error);
    return true; // Optimistically allow write
  }
}

/**
 * Get estimated storage quota
 * @returns {number} Quota in bytes
 */
function getStorageQuota() {
  // Default estimate: 5MB (browsers typically allow 5-10MB)
  return 5 * 1024 * 1024;
}

export { StorageError };
```
