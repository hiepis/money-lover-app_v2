/**
 * Data initialization utilities for first-time setup and migrations
 */

import { storage, StorageError } from './storage.js';
import { initializeDefaultCategories } from './categoryStorage.js';
import { addExpense } from './expenseStorage.js';
import { validateExpense, validateCategory, dataMigration } from './validation.js';

const APP_SETTINGS_KEY = 'app_settings';
const DATA_VERSION_KEY = 'data_version';
const INITIALIZATION_KEY = 'app_initialized';

/**
 * Current data schema version
 */
const CURRENT_DATA_VERSION = '1.0.0';

/**
 * Default app settings
 */
const DEFAULT_APP_SETTINGS = {
  currency: 'USD',
  dateFormat: 'YYYY-MM-DD',
  theme: 'light',
  notifications: true,
  backupReminder: true,
  language: 'en',
  firstTimeSetup: true,
  showIntroTutorial: true,
  autoBackup: false,
  maxStorageUsage: 80, // Percentage
  dataRetentionMonths: 24
};

/**
 * Initialize the app for first-time use
 * @returns {Object} Initialization result
 */
export async function initializeApp() {
  try {
    console.log('Starting app initialization...');
    
    const initResult = {
      success: false,
      isFirstTime: false,
      categoriesCreated: 0,
      sampleDataCreated: 0,
      settingsInitialized: false,
      migrationPerformed: false,
      errors: []
    };

    // Check if app is already initialized
    const isInitialized = storage.get(INITIALIZATION_KEY, false);
    const currentVersion = storage.get(DATA_VERSION_KEY, null);

    if (!isInitialized) {
      console.log('First-time initialization detected');
      initResult.isFirstTime = true;
      
      // Perform first-time setup
      await performFirstTimeSetup(initResult);
    } else if (currentVersion !== CURRENT_DATA_VERSION) {
      console.log(`Data migration needed: ${currentVersion} -> ${CURRENT_DATA_VERSION}`);
      initResult.migrationPerformed = true;
      
      // Perform data migration
      await performDataMigration(currentVersion, initResult);
    } else {
      console.log('App already initialized, checking data integrity...');
      
      // Verify data integrity
      await verifyDataIntegrity(initResult);
    }

    // Mark initialization as complete
    storage.set(INITIALIZATION_KEY, true);
    storage.set(DATA_VERSION_KEY, CURRENT_DATA_VERSION);
    
    initResult.success = true;
    console.log('App initialization completed successfully');
    
    return initResult;
  } catch (error) {
    console.error('App initialization failed:', error);
    return {
      success: false,
      isFirstTime: false,
      categoriesCreated: 0,
      sampleDataCreated: 0,
      settingsInitialized: false,
      migrationPerformed: false,
      errors: [error.message]
    };
  }
}

/**
 * Perform first-time app setup
 * @param {Object} initResult - Initialization result object to update
 */
async function performFirstTimeSetup(initResult) {
  try {
    // Initialize app settings
    const settingsInitialized = initializeAppSettings();
    initResult.settingsInitialized = settingsInitialized;

    // Initialize default categories
    const categories = initializeDefaultCategories();
    initResult.categoriesCreated = categories.length;

    // Initialize sample data for development (optional)
    if (isDevelopmentMode()) {
      const sampleExpenses = await createSampleData();
      initResult.sampleDataCreated = sampleExpenses;
    }

    console.log('First-time setup completed');
  } catch (error) {
    console.error('First-time setup failed:', error);
    initResult.errors.push(`First-time setup: ${error.message}`);
    throw error;
  }
}

/**
 * Initialize app settings with defaults
 * @returns {boolean} Success status
 */
export function initializeAppSettings() {
  try {
    const existingSettings = storage.get(APP_SETTINGS_KEY, {});
    
    // Merge with defaults to ensure all settings exist
    const settings = {
      ...DEFAULT_APP_SETTINGS,
      ...existingSettings,
      lastInitialized: new Date().toISOString()
    };

    const success = storage.set(APP_SETTINGS_KEY, settings);
    if (!success) {
      throw new Error('Failed to save app settings');
    }

    console.log('App settings initialized');
    return true;
  } catch (error) {
    console.error('Failed to initialize app settings:', error);
    return false;
  }
}

/**
 * Get app settings
 * @returns {Object} App settings object
 */
export function getAppSettings() {
  try {
    return storage.get(APP_SETTINGS_KEY, DEFAULT_APP_SETTINGS);
  } catch (error) {
    console.error('Error getting app settings:', error);
    return DEFAULT_APP_SETTINGS;
  }
}

/**
 * Update app settings
 * @param {Object} updates - Settings to update
 * @returns {boolean} Success status
 */
export function updateAppSettings(updates) {
  try {
    const currentSettings = getAppSettings();
    const newSettings = {
      ...currentSettings,
      ...updates,
      lastUpdated: new Date().toISOString()
    };

    const success = storage.set(APP_SETTINGS_KEY, newSettings);
    if (!success) {
      throw new Error('Failed to update app settings');
    }

    return true;
  } catch (error) {
    console.error('Error updating app settings:', error);
    return false;
  }
}

/**
 * Perform data migration from old version to new version
 * @param {string} fromVersion - Current data version
 * @param {Object} initResult - Initialization result object
 */
async function performDataMigration(fromVersion, initResult) {
  try {
    console.log(`Performing data migration from ${fromVersion} to ${CURRENT_DATA_VERSION}`);
    
    // Get all existing data
    const expenses = storage.get('expenses', []);
    const categories = storage.get('categories', []);
    const settings = storage.get(APP_SETTINGS_KEY, {});

    // Perform version-specific migrations
    let migrationSuccess = true;

    if (fromVersion === null || fromVersion === undefined) {
      // Migration from pre-versioned data
      migrationSuccess = await migrateFromUnversioned(expenses, categories, settings);
    } else {
      // Version-specific migrations would go here
      migrationSuccess = await migrateFromVersion(fromVersion, expenses, categories, settings);
    }

    if (!migrationSuccess) {
      throw new Error('Data migration failed');
    }

    console.log('Data migration completed successfully');
  } catch (error) {
    console.error('Data migration failed:', error);
    initResult.errors.push(`Migration: ${error.message}`);
    throw error;
  }
}

/**
 * Migrate from unversioned data format
 * @param {Array} expenses - Existing expenses
 * @param {Array} categories - Existing categories
 * @param {Object} settings - Existing settings
 * @returns {boolean} Success status
 */
async function migrateFromUnversioned(expenses, categories, settings) {
  try {
    // Validate and fix existing data
    const validatedExpenses = [];
    for (const expense of expenses) {
      try {
        const validated = validateExpense(expense);
        validatedExpenses.push(validated);
      } catch (error) {
        console.warn('Skipping invalid expense during migration:', error);
      }
    }

    const validatedCategories = [];
    for (const category of categories) {
      try {
        const validated = validateCategory(category);
        validatedCategories.push(validated);
      } catch (error) {
        console.warn('Skipping invalid category during migration:', error);
      }
    }

    // Ensure default categories exist
    const defaultCategories = initializeDefaultCategories();
    const allCategories = [...validatedCategories, ...defaultCategories];
    
    // Deduplicate categories by ID
    const uniqueCategories = Array.from(
      new Map(allCategories.map(cat => [cat.id, cat])).values()
    );

    // Save migrated data
    storage.set('expenses', validatedExpenses);
    storage.set('categories', uniqueCategories);
    
    // Initialize settings if they don't exist
    if (Object.keys(settings).length === 0) {
      initializeAppSettings();
    }

    return true;
  } catch (error) {
    console.error('Migration from unversioned data failed:', error);
    return false;
  }
}

/**
 * Verify data integrity after initialization
 * @param {Object} initResult - Initialization result object
 */
async function verifyDataIntegrity(initResult) {
  try {
    // Check if essential data exists
    const expenses = storage.get('expenses', []);
    const categories = storage.get('categories', []);
    const settings = storage.get(APP_SETTINGS_KEY, {});

    // Verify categories exist
    if (categories.length === 0) {
      console.log('No categories found, initializing defaults...');
      const defaultCategories = initializeDefaultCategories();
      initResult.categoriesCreated = defaultCategories.length;
    }

    // Verify settings exist
    if (Object.keys(settings).length === 0) {
      console.log('No settings found, initializing defaults...');
      const settingsInitialized = initializeAppSettings();
      initResult.settingsInitialized = settingsInitialized;
    }

    console.log('Data integrity verification completed');
  } catch (error) {
    console.error('Data integrity verification failed:', error);
    initResult.errors.push(`Integrity check: ${error.message}`);
  }
}

/**
 * Reset all app data to defaults
 * @returns {boolean} Success status
 */
export function resetAppData() {
  try {
    console.log('Resetting app data to defaults...');
    
    // Clear all data
    storage.clear(['expenses', 'categories', APP_SETTINGS_KEY, DATA_VERSION_KEY, INITIALIZATION_KEY]);
    
    // Reinitialize
    const initResult = initializeApp();
    
    console.log('App data reset completed');
    return initResult.success;
  } catch (error) {
    console.error('Failed to reset app data:', error);
    return false;
  }
}

/**
 * Export all app data for backup
 * @returns {Object|null} Exported data object
 */
export function exportAppData() {
  try {
    const expenses = storage.get('expenses', []);
    const categories = storage.get('categories', []);
    const settings = storage.get(APP_SETTINGS_KEY, {});

    return {
      version: CURRENT_DATA_VERSION,
      exportDate: new Date().toISOString(),
      data: {
        expenses,
        categories,
        settings
      },
      metadata: {
        expenseCount: expenses.length,
        categoryCount: categories.length,
        dataSize: JSON.stringify({ expenses, categories, settings }).length
      }
    };
  } catch (error) {
    console.error('Failed to export app data:', error);
    return null;
  }
}

/**
 * Import app data from backup
 * @param {Object} backupData - Backup data object
 * @returns {boolean} Success status
 */
export function importAppData(backupData) {
  try {
    if (!dataMigration.validateImportData(backupData)) {
      throw new Error('Invalid backup data format');
    }

    const { data } = backupData;
    
    // Validate data before import
    if (data.expenses) {
      validateExpenses(data.expenses);
    }
    
    if (data.categories) {
      validateCategories(data.categories);
    }

    // Import data
    if (data.expenses) {
      storage.set('expenses', data.expenses);
    }
    
    if (data.categories) {
      storage.set('categories', data.categories);
    }
    
    if (data.settings) {
      storage.set(APP_SETTINGS_KEY, data.settings);
    }

    // Update version and initialization status
    storage.set(DATA_VERSION_KEY, CURRENT_DATA_VERSION);
    storage.set(INITIALIZATION_KEY, true);

    console.log('App data import completed');
    return true;
  } catch (error) {
    console.error('Failed to import app data:', error);
    return false;
  }
}

/**
 * Utility functions for date generation
 */
function getTodaysDate() {
  return new Date().toISOString().split('T')[0];
}

function getYesterdaysDate() {
  const yesterday = new Date();
  yesterday.setDate(yesterday.getDate() - 1);
  return yesterday.toISOString().split('T')[0];
}

function getDateDaysAgo(days) {
  const date = new Date();
  date.setDate(date.getDate() - days);
  return date.toISOString().split('T')[0];
}

/**
 * Check if running in development mode
 * @returns {boolean} Is development mode
 */
function isDevelopmentMode() {
  return import.meta.env?.DEV || process.env?.NODE_ENV === 'development';
}

/**
 * Get initialization status
 * @returns {Object} Initialization status information
 */
export function getInitializationStatus() {
  try {
    return {
      isInitialized: storage.get(INITIALIZATION_KEY, false),
      dataVersion: storage.get(DATA_VERSION_KEY, null),
      currentVersion: CURRENT_DATA_VERSION,
      needsMigration: storage.get(DATA_VERSION_KEY, null) !== CURRENT_DATA_VERSION,
      hasExpenses: storage.get('expenses', []).length > 0,
      hasCategories: storage.get('categories', []).length > 0,
      hasSettings: Object.keys(storage.get(APP_SETTINGS_KEY, {})).length > 0
    };
  } catch (error) {
    console.error('Error getting initialization status:', error);
    return {
      isInitialized: false,
      dataVersion: null,
      currentVersion: CURRENT_DATA_VERSION,
      needsMigration: true,
      hasExpenses: false,
      hasCategories: false,
      hasSettings: false
    };
  }
}
