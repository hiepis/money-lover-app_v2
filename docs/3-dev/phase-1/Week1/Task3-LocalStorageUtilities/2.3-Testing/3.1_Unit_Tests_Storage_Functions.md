```javascript
/**
 * Test environment setup for localStorage utilities

// Mock localStorage for testing
const localStorageMock = {
  store: {},
  
  getItem: function(key) {
    return this.store[key] || null;
  },
  
  setItem: function(key, value) {
    if (this._shouldThrowQuotaError) {
      const error = new Error('QuotaExceededError');
      error.name = 'QuotaExceededError';
      throw error;
    }
    this.store[key] = String(value);
  },
  
  removeItem: function(key) {
    delete this.store[key];
  },
  
  clear: function() {
    this.store = {};
  },
  
  key: function(index) {
    const keys = Object.keys(this.store);
    return keys[index] || null;
  },
  
  get length() {
    return Object.keys(this.store).length;
  },
  
  // Test utilities
  _shouldThrowQuotaError: false,
  _setQuotaError: function(should) {
    this._shouldThrowQuotaError = should;
  },
  
  _getStore: function() {
    return { ...this.store };
  },
  
  _setStore: function(store) {
    this.store = { ...store };
  }
};

// Setup global localStorage mock
global.localStorage = localStorageMock;

// Reset localStorage before each test
beforeEach(() => {
  localStorage.clear();
  localStorage._setQuotaError(false);
});
```

### Core Storage Utility Tests

#### Core Storage Tests (`src/test/storage.test.js`)
```javascript
/**
 * Unit tests for core storage utilities
 */

import { describe, it, expect, beforeEach, vi } from 'vitest';
import { storage, StorageError } from '../utils/storage.js';

describe('Core Storage Utility', () => {
  beforeEach(() => {
    localStorage.clear();
    localStorage._setQuotaError(false);
  });

  describe('storage.get()', () => {
    it('should return null for non-existent key', () => {
      const result = storage.get('nonexistent');
      expect(result).toBeNull();
    });

    it('should return default value for non-existent key', () => {
      const result = storage.get('nonexistent', 'default');
      expect(result).toBe('default');
    });

    it('should parse and return stored JSON data', () => {
      const data = { test: 'value', number: 42 };
      localStorage.setItem('test-key', JSON.stringify(data));
      
      const result = storage.get('test-key');
      expect(result).toEqual(data);
    });

    it('should return default value on JSON parse error', () => {
      localStorage.setItem('bad-json', '{invalid json}');
      
      const result = storage.get('bad-json', 'fallback');
      expect(result).toBe('fallback');
    });

    it('should handle localStorage unavailable', () => {
      const originalGetItem = localStorage.getItem;
      localStorage.getItem = vi.fn(() => {
        throw new Error('localStorage unavailable');
      });

      const result = storage.get('test', 'default');
      expect(result).toBe('default');

      localStorage.getItem = originalGetItem;
    });
  });

  describe('storage.set()', () => {
    it('should store simple data successfully', () => {
      const data = { test: 'value' };
      const result = storage.set('test-key', data);
      
      expect(result).toBe(true);
      expect(localStorage.getItem('test-key')).toBe(JSON.stringify(data));
    });

    it('should store complex nested data', () => {
      const data = {
        array: [1, 2, 3],
        nested: { deep: { value: 'test' } },
        date: new Date().toISOString()
      };
      
      const result = storage.set('complex', data);
      expect(result).toBe(true);
      
      const retrieved = storage.get('complex');
      expect(retrieved).toEqual(data);
    });

    it('should throw StorageError on quota exceeded', () => {
      localStorage._setQuotaError(true);
      
      expect(() => {
        storage.set('test', 'data');
      }).toThrow(StorageError);
    });

    it('should handle localStorage unavailable', () => {
      const originalSetItem = localStorage.setItem;
      localStorage.setItem = vi.fn(() => {
        throw new Error('localStorage unavailable');
      });

      expect(() => {
        storage.set('test', 'data');
      }).toThrow(StorageError);

      localStorage.setItem = originalSetItem;
    });
  });

  describe('storage.remove()', () => {
    it('should remove existing item', () => {
      localStorage.setItem('test', 'value');
      
      const result = storage.remove('test');
      expect(result).toBe(true);
      expect(localStorage.getItem('test')).toBeNull();
    });

    it('should return true even for non-existent item', () => {
      const result = storage.remove('nonexistent');
      expect(result).toBe(true);
    });

    it('should handle localStorage unavailable', () => {
      const originalRemoveItem = localStorage.removeItem;
      localStorage.removeItem = vi.fn(() => {
        throw new Error('localStorage unavailable');
      });

      const result = storage.remove('test');
      expect(result).toBe(false);

      localStorage.removeItem = originalRemoveItem;
    });
  });

  describe('storage.clear()', () => {
    it('should clear specified app keys', () => {
      localStorage.setItem('expenses', '[]');
      localStorage.setItem('categories', '[]');
      localStorage.setItem('other', 'keep');
      
      const result = storage.clear(['expenses', 'categories']);
      expect(result).toBe(true);
      expect(localStorage.getItem('expenses')).toBeNull();
      expect(localStorage.getItem('categories')).toBeNull();
      expect(localStorage.getItem('other')).toBe('keep');
    });

    it('should clear default app keys when none specified', () => {
      localStorage.setItem('expenses', '[]');
      localStorage.setItem('categories', '[]');
      localStorage.setItem('app_settings', '{}');
      localStorage.setItem('data_version', '1.0');
      
      const result = storage.clear();
      expect(result).toBe(true);
      expect(localStorage.getItem('expenses')).toBeNull();
      expect(localStorage.getItem('categories')).toBeNull();
      expect(localStorage.getItem('app_settings')).toBeNull();
      expect(localStorage.getItem('data_version')).toBeNull();
    });
  });

  describe('storage.getSize()', () => {
    it('should calculate storage usage correctly', () => {
      localStorage.setItem('test1', 'value1');
      localStorage.setItem('test2', 'value2');
      
      const size = storage.getSize();
      expect(size.used).toBeGreaterThan(0);
      expect(size.total).toBeGreaterThan(0);
      expect(size.available).toBeGreaterThan(0);
      expect(size.usagePercent).toBeGreaterThanOrEqual(0);
    });

    it('should handle empty storage', () => {
      const size = storage.getSize();
      expect(size.used).toBe(0);
      expect(size.usagePercent).toBe(0);
    });

    it('should handle localStorage unavailable', () => {
      const originalGetItem = localStorage.getItem;
      localStorage.getItem = vi.fn(() => {
        throw new Error('localStorage unavailable');
      });

      const size = storage.getSize();
      expect(size).toEqual({
        total: 0,
        available: 0,
        used: 0,
        usagePercent: 0
      });

      localStorage.getItem = originalGetItem;
    });
  });

  describe('storage.has()', () => {
    it('should return true for existing key', () => {
      localStorage.setItem('test', 'value');
      expect(storage.has('test')).toBe(true);
    });

    it('should return false for non-existent key', () => {
      expect(storage.has('nonexistent')).toBe(false);
    });

    it('should handle localStorage unavailable', () => {
      const originalGetItem = localStorage.getItem;
      localStorage.getItem = vi.fn(() => {
        throw new Error('localStorage unavailable');
      });

      expect(storage.has('test')).toBe(false);

      localStorage.getItem = originalGetItem;
    });
  });

  describe('storage.keys()', () => {
    it('should return all keys when no prefix', () => {
      localStorage.setItem('test1', 'value1');
      localStorage.setItem('test2', 'value2');
      localStorage.setItem('other', 'value3');
      
      const keys = storage.keys();
      expect(keys).toHaveLength(3);
      expect(keys).toContain('test1');
      expect(keys).toContain('test2');
      expect(keys).toContain('other');
    });

    it('should filter keys by prefix', () => {
      localStorage.setItem('app_settings', '{}');
      localStorage.setItem('app_version', '1.0');
      localStorage.setItem('user_data', '[]');
      
      const appKeys = storage.keys('app_');
      expect(appKeys).toHaveLength(2);
      expect(appKeys).toContain('app_settings');
      expect(appKeys).toContain('app_version');
      expect(appKeys).not.toContain('user_data');
    });

    it('should handle localStorage unavailable', () => {
      const originalKey = localStorage.key;
      localStorage.key = vi.fn(() => {
        throw new Error('localStorage unavailable');
      });

      const keys = storage.keys();
      expect(keys).toEqual([]);

      localStorage.key = originalKey;
    });
  });
});

describe('StorageError', () => {
  it('should create error with operation and key', () => {
    const error = new StorageError('Test error', 'set', 'test-key');
    
    expect(error.message).toBe('Test error');
    expect(error.name).toBe('StorageError');
    expect(error.operation).toBe('set');
    expect(error.key).toBe('test-key');
  });

  it('should be instance of Error', () => {
    const error = new StorageError('Test');
    expect(error).toBeInstanceOf(Error);
  });
});
```

### Expense Storage Tests

#### Expense Storage Tests (`src/test/expenseStorage.test.js`)
```javascript
/**
 * Unit tests for expense storage utilities
 */

import { describe, it, expect, beforeEach, vi } from 'vitest';
import {
  getAllExpenses,
  addExpense,
  updateExpense,
  deleteExpense,
  getExpensesByDate,
  getExpensesByMonth,
  getDailyTotal,
  getMonthlyTotal
} from '../utils/expenseStorage.js';

describe('Expense Storage Utilities', () => {
  beforeEach(() => {
    localStorage.clear();
  });

  describe('getAllExpenses()', () => {
    it('should return empty array for no expenses', () => {
      const expenses = getAllExpenses();
      expect(expenses).toEqual([]);
    });

    it('should return expenses sorted by date (newest first)', () => {
      const testExpenses = [
        { id: '1', date: '2024-01-01', amount: 10 },
        { id: '2', date: '2024-01-03', amount: 20 },
        { id: '3', date: '2024-01-02', amount: 15 }
      ];
      
      localStorage.setItem('expenses', JSON.stringify(testExpenses));
      
      const expenses = getAllExpenses();
      expect(expenses).toHaveLength(3);
      expect(expenses[0].date).toBe('2024-01-03'); // Newest first
      expect(expenses[1].date).toBe('2024-01-02');
      expect(expenses[2].date).toBe('2024-01-01');
    });

    it('should handle storage errors gracefully', () => {
      localStorage.setItem('expenses', '{invalid json}');
      
      const expenses = getAllExpenses();
      expect(expenses).toEqual([]);
    });
  });

  describe('addExpense()', () => {
    it('should add valid expense successfully', () => {
      const expenseData = {
        amount: 25.50,
        categoryId: 'food-dining',
        description: 'Lunch',
        date: '2024-01-15'
      };

      const expense = addExpense(expenseData);
      
      expect(expense).toMatchObject(expenseData);
      expect(expense.id).toBeDefined();
      expect(expense.createdAt).toBeDefined();
      expect(expense.updatedAt).toBeDefined();
      
      const allExpenses = getAllExpenses();
      expect(allExpenses).toHaveLength(1);
      expect(allExpenses[0]).toEqual(expense);
    });

    it('should validate expense data', () => {
      const invalidExpense = {
        amount: -10, // Invalid negative amount
        categoryId: 'food-dining',
        date: '2024-01-15'
      };

      expect(() => {
        addExpense(invalidExpense);
      }).toThrow();
    });

    it('should handle storage errors', () => {
      localStorage._setQuotaError(true);
      
      const expenseData = {
        amount: 25.50,
        categoryId: 'food-dining',
        date: '2024-01-15'
      };

      expect(() => {
        addExpense(expenseData);
      }).toThrow();
    });
  });

  describe('updateExpense()', () => {
    it('should update existing expense', () => {
      // Add initial expense
      const initialExpense = addExpense({
        amount: 25.00,
        categoryId: 'food-dining',
        date: '2024-01-15'
      });

      // Update expense
      const updateData = {
        amount: 30.00,
        description: 'Updated description'
      };

      const updatedExpense = updateExpense(initialExpense.id, updateData);
      
      expect(updatedExpense.amount).toBe(30.00);
      expect(updatedExpense.description).toBe('Updated description');
      expect(updatedExpense.categoryId).toBe('food-dining'); // Unchanged
      expect(updatedExpense.id).toBe(initialExpense.id); // Preserved
      expect(updatedExpense.createdAt).toBe(initialExpense.createdAt); // Preserved
      expect(updatedExpense.updatedAt).not.toBe(initialExpense.updatedAt); // Updated
    });

    it('should throw error for non-existent expense', () => {
      expect(() => {
        updateExpense('nonexistent-id', { amount: 10 });
      }).toThrow('Expense with ID nonexistent-id not found');
    });

    it('should validate update data', () => {
      const expense = addExpense({
        amount: 25.00,
        categoryId: 'food-dining',
        date: '2024-01-15'
      });

      expect(() => {
        updateExpense(expense.id, { amount: -10 }); // Invalid amount
      }).toThrow();
    });
  });

  describe('deleteExpense()', () => {
    it('should delete existing expense', () => {
      const expense = addExpense({
        amount: 25.00,
        categoryId: 'food-dining',
        date: '2024-01-15'
      });

      const result = deleteExpense(expense.id);
      expect(result).toBe(true);
      
      const allExpenses = getAllExpenses();
      expect(allExpenses).toHaveLength(0);
    });

    it('should throw error for non-existent expense', () => {
      expect(() => {
        deleteExpense('nonexistent-id');
      }).toThrow('Expense with ID nonexistent-id not found');
    });
  });

  describe('getExpensesByDate()', () => {
    it('should return expenses for specific date', () => {
      addExpense({ amount: 10, categoryId: 'food', date: '2024-01-15' });
      addExpense({ amount: 20, categoryId: 'food', date: '2024-01-16' });
      addExpense({ amount: 30, categoryId: 'food', date: '2024-01-15' });

      const expenses = getExpensesByDate('2024-01-15');
      expect(expenses).toHaveLength(2);
      expenses.forEach(expense => {
        expect(expense.date).toBe('2024-01-15');
      });
    });

    it('should return empty array for date with no expenses', () => {
      const expenses = getExpensesByDate('2024-01-01');
      expect(expenses).toEqual([]);
    });
  });

  describe('getExpensesByMonth()', () => {
    it('should return expenses for specific month', () => {
      addExpense({ amount: 10, categoryId: 'food', date: '2024-01-15' });
      addExpense({ amount: 20, categoryId: 'food', date: '2024-01-20' });
      addExpense({ amount: 30, categoryId: 'food', date: '2024-02-15' });

      const expenses = getExpensesByMonth(2024, 1);
      expect(expenses).toHaveLength(2);
      expenses.forEach(expense => {
        expect(expense.date).toMatch(/^2024-01/);
      });
    });
  });

  describe('getDailyTotal()', () => {
    it('should calculate correct daily total', () => {
      addExpense({ amount: 10.50, categoryId: 'food', date: '2024-01-15' });
      addExpense({ amount: 25.25, categoryId: 'food', date: '2024-01-15' });
      addExpense({ amount: 15.00, categoryId: 'food', date: '2024-01-16' });

      const total = getDailyTotal('2024-01-15');
      expect(total).toBe(35.75);
    });

    it('should return 0 for date with no expenses', () => {
      const total = getDailyTotal('2024-01-01');
      expect(total).toBe(0);
    });
  });

  describe('getMonthlyTotal()', () => {
    it('should calculate correct monthly total', () => {
      addExpense({ amount: 100, categoryId: 'food', date: '2024-01-15' });
      addExpense({ amount: 200, categoryId: 'food', date: '2024-01-20' });
      addExpense({ amount: 50, categoryId: 'food', date: '2024-02-15' });

      const total = getMonthlyTotal(2024, 1);
      expect(total).toBe(300);
    });

    it('should return 0 for month with no expenses', () => {
      const total = getMonthlyTotal(2024, 12);
      expect(total).toBe(0);
    });
  });
});
```
